import cv2

cap = cv2.VideoCapture('video.mp4')

frames = []

while True:
    ret, frame = cap.read()

    if not ret:
        break

    frames.append(frame)

cap.release()

h, w, layers = frames[0].shape

out = cv2.VideoWriter(
    'reverse.mp4',
    cv2.VideoWriter_fourcc(*'mp4v'),
    20, (w, h)
)

for frame in reversed(frames):
    out.write(frame)

out.release()

print("Reverse video created successfully")
