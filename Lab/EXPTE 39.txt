import cv2

cap = cv2.VideoCapture('video.mp4')

frames = []

while True:
    ret, frame = cap.read()

    if not ret:
        break

    frames.append(frame)

fps = cap.get(cv2.CAP_PROP_FPS)

cap.release()

h, w, layers = frames[0].shape

out = cv2.VideoWriter(
    'reverse_slow.mp4',
    cv2.VideoWriter_fourcc(*'mp4v'),
    fps / 2,
    (w, h)
)

for frame in reversed(frames):
    out.write(frame)

out.release()

print("Reverse slow motion video created successfully")
